<script>

  var QUESTIONS = [

    "You wake up feeling tired, even after having an opportunity to sleep.",

    "Small tasks, decisions or responsibilities feel harder than they used to.",

    "You feel irritable, tearful, emotionally numb or less patient than usual.",

    "Your mind keeps worrying, replaying conversations or anticipating problems when you are trying to rest.",

    "You feel responsible for supporting others but have little time or energy for yourself.",

    "You are carrying a loss, change or disappointment that you have not fully acknowledged or had space to process."

  ];


  var BANDS = [

    { title: "Your emotional reserves appear relatively steady.",

      text: "You may still experience demanding moments, but your responses do not suggest a substantial current pattern of emotional exhaustion.",

      next: "Continue protecting your sleep, boundaries and opportunities to recharge.", cls: "band-0" },

    { title: "Your emotional load may be increasing.",

      text: "You might be managing, but some of your emotional reserves appear to be under pressure.",

      next: "Identify one responsibility you can postpone, delegate or simplify this week.", cls: "band-1" },

    { title: "You may be carrying significant emotional exhaustion.",

      text: "Your responses suggest that ongoing pressure, worry, caregiving or unrecognised loss may be affecting your wellbeing.",

      next: "Speak to someone you trust, reduce one avoidable demand and consider what kind of support you need.", cls: "band-2" },

    { title: "Your emotional reserves may be heavily depleted.",

      text: "Your responses suggest that you may need meaningful support and recovery rather than simply trying to push through.",

      next: "Consider arranging an appointment with your GP or a qualified mental health professional, particularly if these difficulties are persistent or affecting daily life.", cls: "band-3" }

  ];


  // Replace this with your real link (quiz / booking page / email)

  var CTA_LINK = "https://griefmasterc-bxpdss53.manus.space/";


  var qWrap = document.getElementById("questions");

  var answers = new Array(QUESTIONS.length).fill(null);


  QUESTIONS.forEach(function (q, i) {

    var qDiv = document.createElement("div");

    qDiv.className = "q";

    var label = document.createElement("div");

    label.className = "q-label";

    label.textContent = (i + 1) + ". " + q;

    qDiv.appendChild(label);

    var opts = document.createElement("div");

    opts.className = "opts";

    [0, 1, 2, 3].forEach(function (v) {

      var o = document.createElement("div");

      o.className = "opt";

      var input = document.createElement("input");

      input.type = "radio"; input.name = "q" + i; input.value = v; input.id = "q" + i + "_" + v;

      input.addEventListener("change", function () { answers[i] = v; checkReady(); });

      var lab = document.createElement("label");

      lab.htmlFor = input.id; lab.textContent = v;

      o.appendChild(input); o.appendChild(lab); opts.appendChild(o);

    });

    qDiv.appendChild(opts);

    qWrap.appendChild(qDiv);

  });


  function checkReady() {

    var all = answers.every(function (a) { return a !== null; });

    document.getElementById("scoreBtn").disabled = !all;

  }


  document.getElementById("scoreBtn").addEventListener("click", function () {

    var total = answers.reduce(function (s, a) { return s + a; }, 0);

    var band = total <= 4 ? 0 : total <= 9 ? 1 : total <= 14 ? 2 : 3;

    var b = BANDS[band];

    document.getElementById("scoreNum").textContent = total;

    var bt = document.getElementById("bandTitle");

    bt.textContent = b.title; bt.className = "band " + b.cls;

    document.getElementById("bandText").textContent = b.text;

    document.getElementById("nextStep").innerHTML = "<b>Your next step:</b> " + b.next;

    document.getElementById("ctaLink").href = CTA_LINK;

    document.getElementById("quizCard").style.display = "none";

    document.getElementById("resultCard").classList.add("show");

    window.scrollTo({ top: 0, behavior: "smooth" });

  });


  document.getElementById("resetBtn").addEventListener("click", function () {

    answers.fill(null);

    document.querySelectorAll("input[type=radio]").forEach(function (r) { r.checked = false; });

    document.getElementById("scoreBtn").disabled = true;

    document.getElementById("resultCard").classList.remove("show");

    document.getElementById("quizCard").style.display = "block";

    window.scrollTo({ top: 0, behavior: "smooth" });

  });

</script>
